"""Run the frozen observer with a one-task plan; this process never loads evaluator data."""

import asyncio
import json
import os
import sys
from pathlib import Path
from unittest.mock import patch

import full_benchmarks as baseline
from browser_use import Browser

folder = Path(sys.argv[1])
job = json.loads((folder / "actor-config.json").read_text())


def verify_runtime():
    import hashlib
    from importlib.metadata import version

    import browser_use

    import lexmount_browser_agent

    expected = job["product"]
    engine = Path(browser_use.__file__).resolve().parent
    package = Path(lexmount_browser_agent.__file__).resolve().parent
    project = package.parents[1]
    actual = {}
    for name, digest in expected["source_sha256"].items():
        if not name.startswith("src/") or not name.endswith(".py"):
            continue
        path = project / name
        actual[name] = hashlib.sha256(path.read_bytes()).hexdigest()
        assert actual[name] == digest, name
    assert engine == package / "engine/browser_use"
    assert version("browser-use") == expected["engine_version"]
    assert "src/lexmount_browser_agent/engine/browser_use/agent/review.py" in actual
    baseline.write(
        folder / "runtime-provenance.json",
        {
            "product_commit": expected["commit"],
            "engine_version": version("browser-use"),
            "engine_source": str(engine),
            "launcher_source": str(package),
            "python_executable": sys.executable,
            "verified_source_sha256": actual,
            "all_matched": True,
        },
    )


verify_runtime()

policy = (
    "Evaluation on public websites. If a site shows an access denial, CAPTCHA, region or login restriction, record the blocker. "
    "Stop requests to that blocked source. Use independent permitted public sources or continue independent requirements. "
    "Do not change identity or routes to regain access: do not hide browser automation flags, spoof a blocked browser identity, "
    "clear or modify challenge cookies, override DNS/IP/Host, probe origins or virtual hosts, or open another browser. "
    "Do not use third-party page-fetch proxies, translation proxies, CAPTCHA solvers or private endpoints to retrieve blocked content. "
    "An independent index or archive that already contains public material is a source; a service that fetches a blocked page on your behalf is a proxy. "
    "Use only this task workspace for local data, its attached browser, and authorized public content. Never access benchmark/evaluator materials, other tasks or system credentials. "
    "Do not contact people, send email, sign third parties up, place real orders or pay; stop before final submission and describe the observed state."
)


async def main():
    from browser_use.agent import loop

    from lexmount_browser_agent import AgentConfig

    source_config = json.load(sys.stdin)
    assert job["feature_config"]["public_source_lookup"] is True
    assert job["feature_config"]["review_answer"] is True
    check_config = AgentConfig(
        **source_config,
        max_steps=job["max_steps"],
        timeout_seconds=job["timeout_seconds"],
        **job["feature_config"],
    )
    assert check_config.public_source_lookup and check_config.review_answer
    assert check_config.max_steps == 100 and check_config.timeout_seconds == 3600
    original_read = baseline.read

    def read(path):
        return source_config if path == baseline.ROOT / ".private/model.json" else original_read(path)

    context = None
    start = Browser.start

    async def start_with_viewport(browser):
        nonlocal context
        result = await start(browser)
        if context is None:
            from bu_lexmount_viewport import FixedViewport

            context = FixedViewport(browser, folder / "viewport.json")
        await context.start()
        return result

    with (
        patch.object(baseline, "read", read),
        patch.object(loop, "SYSTEM_PROMPT", loop.SYSTEM_PROMPT + "\n\n" + policy),
        patch.object(Browser, "start", start_with_viewport if job["fixed_viewport"] else start),
    ):
        try:
            await baseline.worker(
                folder,
                job["max_steps"],
                job["timeout_seconds"],
                {"jobs": [job], "browser_executable": "/opt/google/chrome/chrome"},
            )
        finally:
            if context and context.client:
                try:
                    await context.close()
                except Exception as error:  # noqa: BLE001 - capture cleanup must preserve the task result
                    print("viewport cleanup failed", type(error).__name__, flush=True)
    result = original_read(folder / "execution.json")
    config = original_read(folder / "agent/run.json")["config"]
    assert config["review_answer"] is True and config["public_source_lookup"] is True
    turns = result.get("model_turns")
    assert turns is None or (isinstance(turns, int) and turns <= job["max_steps"] + 1)
    review = folder / "agent/workspace/.browser-use/answer-review.json"
    baseline.write(
        folder / "request-accounting.json",
        {
            "product_commit": job["product"]["commit"],
            "model_turns_including_answer_review": turns,
            "model_http_requests_including_answer_review": result.get("model_http_requests"),
            "usage_including_answer_review": result.get("usage"),
            "cost_usd_including_answer_review": result.get("cost_usd"),
            "review_receipt": str(review) if review.exists() else None,
            "review_status": original_read(review).get("status") if review.exists() else "not_run",
            "note": "The engine counts reviewer requests in the shared task request budget and its reported model usage/cost. HTTP retries and compaction can make transport request count differ from model_turns.",
        },
    )


if __name__ == "__main__":
    if os.environ.get("SEMANTIC_PROBE_ONLY") == "1":

            config = AgentConfig(
                **json.load(sys.stdin),

                public_source_lookup=True,

            )
            task = (
                "Read this local page and return the names and costs of entries costing less than 10. "
                "Keep entries with missing cost separately as unverified. "
                "Page: data:text/html,<title>Local%20inventory</title>"
                "<table><tr><th>Name</th><th>Cost</th></tr>"
                "<tr><td>Alpha</td><td>5</td></tr><tr><td>Beta</td><td>20</td></tr>"
                "<tr><td>Gamma</td><td>unknown</td></tr></table>"
            )
            result = await run_task(task, config, folder / "model-check")
            path = folder / "model-check/workspace/.browser-use/answer-review.json"
            assert result["model_turns"] <= 13
            assert path.is_file(), "Review did not run; inspect the local probe receipts"
            baseline.write(
                folder / "semantic-probe.json",
                {
                    "product_commit": job["product"]["commit"],
                    "task_origin": "local synthetic inventory",
                    "benchmark_tasks": 0,
                    "model_turns": result.get("model_turns"),
                    "usage": result.get("usage"),
                    "cost_usd": result.get("cost_usd"),
                    "status": result.get("status"),
                    "review_status": baseline.read(path).get("status"),
                },
            )
            print("local semantic probe receipts saved", flush=True)

        asyncio.run(semantic_probe())
    elif os.environ.get("PROBE_ONLY") == "1":

        async def probe():
            from browser_use.agent import loop

            from lexmount_browser_agent import AgentConfig

            assert hasattr(loop, "SYSTEM_PROMPT")
            from browser_use.repl import Repl

            browser = Browser("chromium", headless=True, executable_path="/opt/google/chrome/chrome")
            repl = None
            try:
                env = await browser.start()
                repl = Repl(folder / "work", env, mode="ultrafast", public_lookup=True)
                out = await repl.call(
                    "exec",
                    "goto_url('data:text/html,<title>Probe</title><p>example row</p>'); wait_for_element('p'); print(find_text('example') if 'find_text' in globals() else page_info())",
                    45,
                )
                print(out["error"] or out["output"], flush=True)
                assert not out["error"] and "Probe" in out["output"]
                assert not (baseline.ROOT / ".private/model.json").exists()
                assert not (baseline.ROOT / "full-600-20261114").exists()
                check = await repl.call("exec", "print('save_answer' in globals())", 20)
                assert ("True" if os.environ["EXPECTED_ARM"] == "candidate" else "False") in check[
                    "output"
                ], check
                check = await repl.call(
                    "exec",
                    "print(read_page('example')['passages']); print(recall('example')['matching_cells'])",
                    30,
                )
                assert check["error"] is None and "example row" in check["output"], check["error"]
                check = await repl.call(
                    "exec",
                    """rows=[{'name':'Alpha','cost':5},{'name':'Beta','cost':20},{'name':'Gamma'}]
selected=check_candidates({'cost':lambda v:v<10},rows,key='name')
assert selected['matched']==['Alpha'] and selected['pending']=={'Gamma':['cost']}
save_answer('Observed Alpha')
print('local tools passed')""",
                    20,
                )
                assert check["error"] is None and "local tools passed" in check["output"]
                config = AgentConfig(base_url="https://unused.invalid/v1", api_key="unused")
                assert config.review_answer and config.public_source_lookup
                from browser_use.agent.review import AnswerReview, install_answer_review

                assert callable(install_answer_review) and AnswerReview(feedback=None).feedback is None
                assert not list((folder / "model_calls").glob("*.request.json"))
                print("isolated browser probe passed", os.environ["EXPECTED_ARM"], flush=True)
            finally:
                if repl:
                    await repl.close()
                await browser.stop()

        asyncio.run(probe())
    else:
        asyncio.run(main())
